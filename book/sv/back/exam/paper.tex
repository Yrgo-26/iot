\section*{Del A --- det egna protokollet (12 p)}

Checksumman är en 16-bitars summering av \textbf{alla bytes fram till CHK-fältet}. Fält större än
en byte (SOF, SEQ, CHK) skickas big-endian.

\begin{console}
SOF (2) | LEN (1) | TYPE (1) | DST (1) | SRC (1) | SEQ (2) | DATA (N) | CHK (2)
\end{console}

\noindent \code{SOF = 0xA5F7}. Frametyper: \code{Ping = 0x00}, \code{Pong = 0x01},
\code{StatusRequest = 0x02}, \code{StatusResponse = 0x03}, \code{Ack = 0x04}, \code{Nack = 0x05}.

\exercise{Bygg två frames}{4 p}
\label{ex:d:1}
\begin{enumerate}[a)]
  \item En nod med adress \code{0x17} skickar en \code{StatusRequest} till en sensor med adress
    \code{0x25}. Sekvensnumret är \code{0x7F05}. Skriv upp hela framen byte för byte. (2 p)
  \item Sensorn svarar med ett \code{StatusResponse} innehållande temperaturen \code{0x16}
    (en byte). Skriv upp den framen. (2 p)
\end{enumerate}

\exercise{Frame-parsern}{2 p}
\label{ex:d:2}
\begin{enumerate}[a)]
  \item Beskriv kort vad parsern gör i \code{WaitForSof1} respektive \code{WaitForSof2}. (1 p)
  \item Parsern står i \code{WaitForSof2} och tar emot en byte som inte är \code{0xF7}. Vad ska
    den göra? Det finns ett svar som är nästan rätt och ett som är rätt --- motivera vilket som är
    vilket. (1 p)
\end{enumerate}

\exercise{Buss och routing}{2 p}
\label{ex:d:3}
\begin{enumerate}[a)]
  \item Sker routingen i bussen eller i noden? Motivera. (1 p)
  \item En nod tar emot en frame med korrekt checksumma, men \code{DST} är inte nodens adress.
    Vad ska noden göra, och varför ska den \emph{inte} skicka \code{Nack}? (1 p)
\end{enumerate}

\exercise{Tillförlitlighet}{2 p}
\label{ex:d:4}
\begin{enumerate}[a)]
  \item Nod A skickar en \code{StatusRequest} och får ingen \code{Ack}. Beskriv flödet med timeout
    och omsändning. Ska \code{SEQ} vara samma eller nytt, och varför? (1 p)
  \item \code{Ack}-framen tappas på vägen tillbaka, så A skickar om. B tar emot samma frame en
    andra gång. Vad ska B göra, och vad ska B \emph{inte} göra? (1 p)
\end{enumerate}

\exercise{Byte-ström och felmodell}{2 p}
\label{ex:d:5}
En nod tar emot:
\begin{console}
32 74 00 FF A5 F7 03 03 25 17 7F 05 10 20 30 02 C2
\end{console}
\begin{enumerate}[a)]
  \item Vid vilket index börjar framen, och vad ska parsern göra med datan före? (1 p)
  \item Antag i stället att byten \code{0x20} tappas. Vad händer med tolkningen, och i vilket
    tillstånd hamnar parsern? (1 p)
\end{enumerate}

\section*{Del B --- CAN och drivrutinen (13 p)}

\exercise{CAN-bussen}{3 p}
\label{ex:d:6}
\begin{enumerate}[a)]
  \item Vad är ett CAN-ID, och vilka \textbf{två} roller har det samtidigt? (1 p)
  \item Två noder börjar sända samtidigt. Nod A har \code{0x100}, nod B har \code{0x0FF}. Vilken
    vinner, vad gör förloraren, och varför krävs ingen bussmästare? (1 p)
  \item Räkna upp \textbf{tre} mekanismer från del A som du inte behövde implementera i mjukvara,
    och säg var i systemet de utförs i stället. (1 p)
\end{enumerate}

\exercise{Registerkartan}{4 p}
\label{ex:d:7}
\begin{enumerate}[a)]
  \item Du ska skicka en frame med \code{id = 0x2A7}, \code{dlc = 5} och databytesen
    \code{01 02 03 04 05}. Skriv upp de fem registervärdena som ska skrivas, i rätt ordning. Varför
    spelar ordningen roll? (2 p)
  \item Du läser \code{RX_ID = 0x00000481}, \code{RX_DLC = 0x00000003},
    \code{RX_DATA_LO = 0xDEADBE00}, \code{RX_DATA_HI = 0x00000000}. Vilken identifierare kom framen
    från, vilka databytes innehåller den, och vad ska \code{frame.data[3]} till
    \code{frame.data[7]} innehålla efter \code{receive()}? (1 p)
  \item Vilket register måste skrivas innan nästa frame kan tas emot, och med vilket värde? Vad
    händer om drivrutinen glömmer det? (1 p)
\end{enumerate}

\exercise{SPI-transaktionen}{3 p}
\label{ex:d:8}
Följande bytes går ut på \code{MOSI}. För var och en: läsning eller skrivning, vilket register, och
vilket värde? (2 p)
\begin{console}
a)  81 00 00 07 FF
b)  85 00 00 00 01
c)  8B 00 00 00 00
d)  C2 00 00 00 03
\end{console}
\begin{enumerate}[e)]
  \item Skriv upp den kompletta femBytestransaktionen för att \textbf{läsa} \code{RX_DLC}. (1 p)
\end{enumerate}

\exercise{Klibbiga bitar och begränsningar}{3 p}
\label{ex:d:9}
\begin{enumerate}[a)]
  \item Kontrollern signalerar med pulser på en klockcykel, 20~ns vid 50~MHz. En SPI-transaktion
    tar omkring 40~µs. Förklara varför registerbanken behövs, och varför \code{STATUS} bit 0 måste
    nollställas av en \emph{skrivning} i stället för automatiskt vid läsning. (1 p)
  \item Din nod sänder en frame och förlorar arbitreringen. \code{STATUS} bit 0 kommer tillbaka
    precis som vid en lyckad sändning. Vad ska drivrutinen göra för att ta reda på hur det gick,
    och vad kan den fortfarande \emph{inte} ta reda på? (1 p)
  \item \code{clearError()} skriver \code{0x0} till \code{ERROR_FLAGS}, medan \code{receive()}
    skriver \code{0x1} till \code{RX_ACK}. En student tycker att det är inkonsekvent och skriver
    \code{0x1} i båda. Vad blir symptomet, och när upptäcks det? (1 p)
\end{enumerate}
