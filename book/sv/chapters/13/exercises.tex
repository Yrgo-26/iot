\section*{Övningar}
\addcontentsline{toc}{section}{Övningar}

\exercise{Stegen}{Resonemang}
\label{ex:13:1}
\begin{enumerate}[a)]
  \item Vad bevisar steg 2 som steg 1 inte gjorde?
  \item Varför räcker det inte att läsa tillbaka samma värde du skrev?
  \item Vilket steg är det enda testet av \code{AvrSpiTransport}, och varför finns inget
    automattest för den klassen?
\end{enumerate}

\exercise{Från symptom till hypotes}{Resonemang}
\label{ex:13:2}
Du läser \code{STATUS} och får \code{0x00000000} varje gång.
\begin{enumerate}[a)]
  \item Räkna upp minst fyra möjliga orsaker.
  \item Ordna dem efter hur snabbt de går att utesluta.
  \item Vilken mätning skiljer de två första åt?
\end{enumerate}

\exercise{Vems fel?}{Resonemang}
\label{ex:13:3}
Din nod skickar en frame, men ingenting syns på bussen.
\begin{enumerate}[a)]
  \item Ligger felet i din drivrutin, i FPGA:n eller i kopplingen?
  \item Hur avgör du det \emph{utan} att gissa?
  \item Formulera den felrapport du skulle skicka till hårdvarugruppen, med det du vet så här
    långt.
\end{enumerate}

\exercise{Termineringen}{Räkning}
\label{ex:13:4}
Du mäter resistansen över den strömlösa bussen.
\begin{enumerate}[a)]
  \item Vad ska du få med korrekt terminering?
  \item Du får 40~\textohm. Vad är fel?
  \item Du får 120~\textohm. Vad är fel, och när skulle det visa sig?
\end{enumerate}
